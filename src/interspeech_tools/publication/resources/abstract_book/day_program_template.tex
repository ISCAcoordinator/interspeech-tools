\BLOCK{ for session in sessions }
\sessionslot{\VAR{session[0].id}}
            {\VAR{session[0].name.replace("Special Session - ", "") | compact_numbers}}
            {\VAR{session[0].start | extract_verbose_date}}
            {\VAR{session[1] | format_timeslot}}
            {\VAR{session[2].name}}
            {\VAR{session[0] | type_session}}
            {}

  \BLOCK{ for presentation in session[0].presentations }
    \BLOCK{ if presentation | is_survey }
\survey{\VAR{presentation.id}}
       {\VAR{presentation.title | latexify}}
       {\VAR{presentation | presenters}}
       {\VAR{session[0] | get_presentation_timeslot(loop.index-1)}}

    \BLOCK{ elif presentation | is_bluesky }

\bluesky{\VAR{presentation.id}}
       {\VAR{presentation.title | latexify}}
       {\VAR{presentation | presenters}}
       {\VAR{session[0] | get_presentation_timeslot(loop.index-1)}}
       {\VAR{presentation.presented_paper.abstract | latexify}}

    \BLOCK{ elif presentation | has_paper }

\presentation{\VAR{presentation | type_presentation}}
             {\VAR{presentation.id}}
             {\VAR{presentation.presented_paper.title | latexify}}
             {\VAR{presentation | authors}}
             {\VAR{session[0] | get_presentation_timeslot(loop.index-1)}}
             {\VAR{presentation.presented_paper.abstract | latexify}}

    \BLOCK{ endif }
  \BLOCK{ endfor }


\BLOCK{ endfor }
