\BLOCK{ for session in sessions }
\sessionslot{\VAR{session.id}}
            {\VAR{session.name | compact_numbers}}
            {\VAR{session.start | extract_verbose_date}}
            {\VAR{session | get_session_timeslot}}
            {\VAR{session.room.name}}
            {\VAR{session | type_session}}
            {}

  \BLOCK{ for presentation in session.presentations }
    \BLOCK{ if presentation | is_survey }
\survey{\VAR{presentation.id}}
       {\VAR{presentation.title | latexify}}
       {\VAR{presentation | presenters}}
       {\VAR{session | get_presentation_timeslot(loop.index-1)}}

    \BLOCK{ elif presentation | is_bluesky }

\bluesky{\VAR{presentation.presented_paper.id}}
       {\VAR{presentation.title | latexify}}
       {\VAR{presentation | presenters}}
       {\VAR{session | get_presentation_timeslot(loop.index-1)}}
       {\VAR{presentation.presented_paper.abstract | latexify}}

    \BLOCK{ elif presentation | has_paper }

\presentation{\VAR{presentation | type_presentation}}
             {\VAR{presentation.presented_paper.id}}
             {\VAR{presentation.presented_paper.title | latexify}}
             {\VAR{presentation | authors}}
             {\VAR{session | get_presentation_timeslot(loop.index-1)}}
             {\VAR{presentation.presented_paper.abstract | latexify}}

    \BLOCK{ endif }
  \BLOCK{ endfor }


\BLOCK{ endfor }
